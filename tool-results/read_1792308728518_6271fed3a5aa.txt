     1→'use client'
     2→import { useState, useRef } from 'react'
     3→import { useRouter } from 'next/navigation'
     4→import { Card } from '@/components/ui/card'
     5→import { Button } from '@/components/ui/button'
     6→import { Input } from '@/components/ui/input'
     7→import { Label } from '@/components/ui/label'
     8→import { Textarea } from '@/components/ui/textarea'
     9→import { Switch } from '@/components/ui/switch'
    10→import { Badge } from '@/components/ui/badge'
    11→import {
    12→  Dialog, DialogContent, DialogTrigger, DialogHeader, DialogTitle
    13→} from '@/components/ui/dialog'
    14→import { toast } from '@/lib/toast'
    15→import { FileUpload } from '@/components/shared/file-upload'
    16→import {
    17→  Plus, Trash2, ArrowUp, ArrowDown, Eye, EyeOff, Save,
    18→  Image as ImageIcon, Type, Mail, Gift, Layout, Scroll, Grid3x3, Code,
    19→  GripVertical, Monitor, Smartphone, X, Settings2, Copy
    20→} from 'lucide-react'

    22→const SECTION_TYPES = [
    23→  { type: 'hero', label: 'Hero Banner', icon: Layout, description: 'Large image with title, subtitle, and CTA button' },
    24→  { type: 'product-grid', label: 'Product Grid', icon: Grid3x3, description: 'Show featured, new, or filtered products' },
    25→  { type: 'marquee', label: 'Marquee', icon: Scroll, description: 'Scrolling text banner — great for announcements' },
    26→  { type: 'text', label: 'Text Block', icon: Type, description: 'Heading + paragraph of custom text' },
    27→  { type: 'image', label: 'Image', icon: ImageIcon, description: 'Single image with optional link' },
    28→  { type: 'newsletter', label: 'Newsletter Signup', icon: Mail, description: 'Email subscribe form with prompt text' },
    29→  { type: 'discounts', label: 'Discount Codes', icon: Gift, description: 'Show active discount codes customers can copy' },
    30→  { type: 'custom-html', label: 'Custom HTML', icon: Code, description: 'Raw HTML code block' }
    31→]
    32→
    33→export function PageBuilder({ initialSections, categories }) {
    34→  const router = useRouter()
    35→  const [sections, setSections] = useState(initialSections)
    36→  const [adding, setAdding] = useState(false)
    37→  const [saving, setSaving] = useState(null)
    38→  const [selectedSection, setSelectedSection] = useState(null)  // section id for side panel
    39→  const [previewDevice, setPreviewDevice] = useState('desktop')  // desktop | mobile
    40→  const [draggedId, setDraggedId] = useState(null)
    41→  const [dragOverId, setDragOverId] = useState(null)
    42→
    43→  const addSection = async (type) => {
    44→    const newSection = {
    45→      type,
    46→      config: defaultConfig(type),
    47→      sortOrder: sections.length,
    48→      isVisible: true
    49→    }
    50→    try {
    51→      const r = await fetch('/api/admin/sections', {
    52→        method: 'POST',
    53→        headers: { 'Content-Type': 'application/json' },
    54→        body: JSON.stringify({ pageType: 'home', ...newSection })
    55→      })
    56→      if (r.ok) {
    57→        const d = await r.json()
    58→        const newSec = { ...d.section, config: JSON.parse(d.section.config) }
    59→        setSections([...sections, newSec])
    60→        toast.success('Section added')
    61→        setAdding(false)
    62→        setSelectedSection(newSec.id)
    63→        router.refresh()
    64→      } else {
    65→        throw new Error('Failed')
    66→      }
    67→    } catch (e) {
    68→      toast.error('Failed', e.message)
    69→    }
    70→  }
    71→
    72→  const updateSection = (id, config) => {
    73→    setSections(sections.map(s => s.id === id ? { ...s, config } : s))
    74→  }
    75→
    76→  const saveSection = async (id) => {
    77→    setSaving(id)
    78→    const section = sections.find(s => s.id === id)
    79→    try {
    80→      const r = await fetch(`/api/admin/sections/${id}`, {
    81→        method: 'PATCH',
    82→        headers: { 'Content-Type': 'application/json' },
    83→        body: JSON.stringify({
    84→          config: JSON.stringify(section.config),
    85→          isVisible: section.isVisible
    86→        })
    87→      })
    88→      if (r.ok) {
    89→        toast.success('Saved')
    90→        router.refresh()
    91→      }
    92→    } catch (e) {
    93→      toast.error('Failed', e.message)
    94→    } finally {
    95→      setSaving(null)
    96→    }
    97→  }
    98→
    99→  const toggleVisible = async (id) => {
   100→    const section = sections.find(s => s.id === id)
   101→    const updated = { ...section, isVisible: !section.isVisible }
   102→    setSections(sections.map(s => s.id === id ? updated : s))
   103→    await saveSection(id)
   104→  }
   105→
   106→  const duplicateSection = async (id) => {
   107→    const section = sections.find(s => s.id === id)
   108→    if (!section) return
   109→    try {
   110→      const r = await fetch('/api/admin/sections', {
   111→        method: 'POST',
   112→        headers: { 'Content-Type': 'application/json' },
   113→        body: JSON.stringify({
   114→          pageType: 'home',
   115→          sectionType: section.sectionType,
   116→          config: JSON.stringify(section.config),
   117→          sortOrder: sections.length,
   118→          isVisible: true
   119→        })
   120→      })
   121→      if (r.ok) {
   122→        const d = await r.json()
   123→        const newSec = { ...d.section, config: JSON.parse(d.section.config) }
   124→        setSections([...sections, newSec])
   125→        toast.success('Section duplicated')
   126→        router.refresh()
   127→      }
   128→    } catch (e) {
   129→      toast.error('Failed', e.message)
   130→    }
   131→  }
   132→
   133→  const moveSection = (id, direction) => {
   134→    const idx = sections.findIndex(s => s.id === id)
   135→    const newIdx = direction === 'up' ? idx - 1 : idx + 1
   136→    if (newIdx < 0 || newIdx >= sections.length) return
   137→    const reordered = [...sections]
   138→    const [moved] = reordered.splice(idx, 1)
   139→    reordered.splice(newIdx, 0, moved)
   140→    const withOrders = reordered.map((s, i) => ({ ...s, sortOrder: i }))
   141→    setSections(withOrders)
   142→    // Persist all sort orders
   143→    Promise.all(withOrders.map(s =>
   144→      fetch(`/api/admin/sections/${s.id}`, {
   145→        method: 'PATCH',
   146→        headers: { 'Content-Type': 'application/json' },
   147→        body: JSON.stringify({ sortOrder: s.sortOrder })
   148→      })
   149→    )).then(() => {
   150→      toast.success('Reordered')
   151→      router.refresh()
   152→    })
   153→  }
   154→
   155→  // Drag and drop
   156→  const handleDragStart = (e, id) => {
   157→    setDraggedId(id)
   158→    e.dataTransfer.effectAllowed = 'move'
   159→  }
   160→
   161→  const handleDragOver = (e, id) => {
   162→    e.preventDefault()
   163→    e.dataTransfer.dropEffect = 'move'
   164→    if (id !== draggedId) setDragOverId(id)
   165→  }
   166→
   167→  const handleDrop = (e, targetId) => {
   168→    e.preventDefault()
   169→    if (!draggedId || draggedId === targetId) return
   170→    const draggedIdx = sections.findIndex(s => s.id === draggedId)
   171→    const targetIdx = sections.findIndex(s => s.id === targetId)
   172→    if (draggedIdx === -1 || targetIdx === -1) return
   173→    const reordered = [...sections]
   174→    const [moved] = reordered.splice(draggedIdx, 1)
   175→    reordered.splice(targetIdx, 0, moved)
   176→    const withOrders = reordered.map((s, i) => ({ ...s, sortOrder: i }))
   177→    setSections(withOrders)
   178→    setDraggedId(null)
   179→    setDragOverId(null)
   180→    // Persist
   181→    Promise.all(withOrders.map(s =>
   182→      fetch(`/api/admin/sections/${s.id}`, {
   183→        method: 'PATCH',
   184→        headers: { 'Content-Type': 'application/json' },
   185→        body: JSON.stringify({ sortOrder: s.sortOrder })
   186→      })
   187→    )).then(() => {
   188→      toast.success('Reordered')
   189→      router.refresh()
   190→    })
   191→  }
   192→
   193→  const delSection = async (id) => {
   194→    if (!confirm('Delete this section?')) return
   195→    const r = await fetch(`/api/admin/sections/${id}`, { method: 'DELETE' })
   196→    if (r.ok) {
   197→      setSections(sections.filter(s => s.id !== id))
   198→      if (selectedSection === id) setSelectedSection(null)
   199→      toast.success('Deleted')
   200→      router.refresh()
   201→    }
   202→  }
   203→
   204→  const selectedSectionData = sections.find(s => s.id === selectedSection)
   205→
   206→  return (
   207→    <div className="flex gap-6">
   208→      {/* ===== Left: Section list + preview ===== */}
   209→      <div className="flex-1 min-w-0 space-y-4">
   210→        {/* Toolbar */}
   211→        <div className="flex items-center justify-between gap-3">
   212→          <div className="flex items-center gap-2">
   213→            <div className="inline-flex rounded-md border border-border overflow-hidden">
   214→              <button
   215→                onClick={() => setPreviewDevice('desktop')}
   216→                className={`px-3 py-1.5 text-xs font-medium flex items-center gap-1.5 ${previewDevice === 'desktop' ? 'bg-foreground text-background' : 'hover:bg-muted'}`}
   217→              >
   218→                <Monitor className="h-3.5 w-3.5" /> Desktop
   219→              </button>
   220→              <button
   221→                onClick={() => setPreviewDevice('mobile')}
   222→                className={`px-3 py-1.5 text-xs font-medium flex items-center gap-1.5 ${previewDevice === 'mobile' ? 'bg-foreground text-background' : 'hover:bg-muted'}`}
   223→              >
   224→                <Smartphone className="h-3.5 w-3.5" /> Mobile
   225→              </button>
   226→            </div>
   227→          </div>
   228→          <Dialog open={adding} onOpenChange={setAdding}>
   229→            <DialogTrigger asChild>
   230→              <Button size="sm">
   231→                <Plus className="h-4 w-4 mr-1.5" /> Add Section
   232→              </Button>
   233→            </DialogTrigger>
   234→            <DialogContent className="max-w-2xl">
   235→              <DialogHeader>
   236→                <DialogTitle>Add a section</DialogTitle>
   237→              </DialogHeader>
   238→              <div className="grid sm:grid-cols-2 gap-2">
   239→                {SECTION_TYPES.map(t => (
   240→                  <button
   241→                    key={t.type}
   242→                    onClick={() => addSection(t.type)}
   243→                    className="text-left p-4 rounded-md border border-border hover:border-foreground/40 hover:bg-muted/30 transition-colors group"
   244→                  >
   245→                    <div className="flex items-start gap-3">
   246→                      <div className="h-9 w-9 rounded-lg bg-muted group-hover:bg-foreground group-hover:text-background flex items-center justify-center transition-colors">
   247→                        <t.icon className="h-4 w-4" />
   248→                      </div>
   249→                      <div>
   250→                        <p className="text-sm font-medium">{t.label}</p>
   251→                        <p className="text-xs text-muted-foreground mt-0.5">{t.description}</p>
   252→                      </div>
   253→                    </div>
   254→                  </button>
   255→                ))}
   256→              </div>
   257→            </DialogContent>
   258→          </Dialog>
   259→        </div>
   260→
   261→        {/* Preview canvas */}
   262→        <div className={`mx-auto transition-all ${previewDevice === 'mobile' ? 'max-w-[375px]' : 'max-w-full'}`}>
   263→          <div className="rounded-xl border-2 border-dashed border-border overflow-hidden bg-background">
   264→            {sections.length === 0 ? (
   265→              <div className="p-16 text-center">
   266→                <Layout className="h-10 w-10 text-muted-foreground/50 mx-auto mb-3" />
   267→                <h3 className="font-semibold">Your page is empty</h3>
   268→                <p className="text-sm text-muted-foreground mt-1 max-w-sm mx-auto">
   269→                  Add sections to build your home page. Drag to reorder, click to edit.
   270→                </p>
   271→                <Button onClick={() => setAdding(true)} className="mt-4">
   272→                  <Plus className="h-4 w-4 mr-2" /> Add your first section
   273→                </Button>
   274→              </div>
   275→            ) : (
   276→              <div className="divide-y divide-border">
   277→                {sections.map((section, idx) => {
   278→                  const sectionType = SECTION_TYPES.find(t => t.type === section.sectionType)
   279→                  const Icon = sectionType?.icon || Code
   280→                  const isSelected = selectedSection === section.id
   281→                  const isDragOver = dragOverId === section.id
   282→                  return (
   283→                    <div
   284→                      key={section.id}
   285→                      draggable
   286→                      onDragStart={(e) => handleDragStart(e, section.id)}
   287→                      onDragOver={(e) => handleDragOver(e, section.id)}
   288→                      onDrop={(e) => handleDrop(e, section.id)}
   289→                      onDragEnd={() => { setDraggedId(null); setDragOverId(null) }}
   290→                      className={`relative group transition-all ${
   291→                        isSelected ? 'ring-2 ring-foreground ring-inset' : ''
   292→                      } ${
   293→                        isDragOver ? 'border-t-4 border-t-foreground' : ''
   294→                      } ${
   295→                        !section.isVisible ? 'opacity-50' : ''
   296→                      }`}
   297→                    >
   298→                      {/* Section header bar */}
   299→                      <div className="flex items-center justify-between px-3 py-2 bg-muted/30 border-b border-border">
   300→                        <div className="flex items-center gap-2">
   301→                          <GripVertical className="h-4 w-4 text-muted-foreground cursor-grab active:cursor-grabbing" />
   302→                          <Icon className="h-3.5 w-3.5 text-muted-foreground" />
   303→                          <span className="text-xs font-medium">{sectionType?.label || section.sectionType}</span>
   304→                          {!section.isVisible && <Badge variant="secondary" className="text-[10px]">Hidden</Badge>}
   305→                        </div>
   306→                        <div className="flex items-center gap-0.5 opacity-0 group-hover:opacity-100 transition-opacity">
   307→                          <button onClick={() => moveSection(section.id, 'up')} disabled={idx === 0} className="p-1 hover:bg-muted rounded text-muted-foreground hover:text-foreground disabled:opacity-30" aria-label="Move up">
   308→                            <ArrowUp className="h-3.5 w-3.5" />
   309→                          </button>
   310→                          <button onClick={() => moveSection(section.id, 'down')} disabled={idx === sections.length - 1} className="p-1 hover:bg-muted rounded text-muted-foreground hover:text-foreground disabled:opacity-30" aria-label="Move down">
   311→                            <ArrowDown className="h-3.5 w-3.5" />
   312→                          </button>
   313→                          <button onClick={() => duplicateSection(section.id)} className="p-1 hover:bg-muted rounded text-muted-foreground hover:text-foreground" aria-label="Duplicate">
   314→                            <Copy className="h-3.5 w-3.5" />
   315→                          </button>
   316→                          <button onClick={() => toggleVisible(section.id)} className="p-1 hover:bg-muted rounded text-muted-foreground hover:text-foreground" aria-label="Toggle visibility">
   317→                            {section.isVisible ? <Eye className="h-3.5 w-3.5" /> : <EyeOff className="h-3.5 w-3.5" />}
   318→                          </button>
   319→                          <button onClick={() => delSection(section.id)} className="p-1 hover:bg-muted rounded text-muted-foreground hover:text-destructive" aria-label="Delete">
   320→                            <Trash2 className="h-3.5 w-3.5" />
   321→                          </button>
   322→                        </div>
   323→                      </div>
   324→
   325→                      {/* Live preview */}
   326→                      <div
   327→                        className="cursor-pointer"
   328→                        onClick={() => setSelectedSection(section.id)}
   329→                      >
   330→                        <SectionPreview section={section} />
   331→                      </div>
   332→                    </div>
   333→                  )
   334→                })}
   335→              </div>
   336→            )}
   337→          </div>
   338→        </div>
   339→
   340→        <p className="text-xs text-muted-foreground text-center">
   341→          💡 Drag sections to reorder · Click a section to edit its content · Use the eye icon to hide temporarily
   342→        </p>
   343→      </div>
   344→
   345→      {/* ===== Right: Side panel for editing selected section ===== */}
   346→      {selectedSectionData && (
   347→        <div className="w-80 shrink-0">
   348→          <div className="sticky top-24">
   349→            <Card className="p-0">
   350→              {/* Panel header */}
   351→              <div className="flex items-center justify-between px-4 py-3 border-b border-border">
   352→                <div className="flex items-center gap-2">
   353→                  <Settings2 className="h-4 w-4" />
   354→                  <h3 className="text-sm font-semibold">Edit Section</h3>
   355→                </div>
   356→                <button onClick={() => setSelectedSection(null)} className="text-muted-foreground hover:text-foreground">
   357→                  <X className="h-4 w-4" />
   358→                </button>
   359→              </div>
   360→
   361→              {/* Panel content */}
   362→              <div className="p-4 max-h-[60vh] overflow-y-auto">
   363→                <SectionConfigEditor
   364→                  section={selectedSectionData}
   365→                  categories={categories}
   366→                  onUpdate={(config) => updateSection(selectedSectionData.id, config)}
   367→                />
   368→              </div>
   369→
   370→              {/* Panel footer */}
   371→              <div className="px-4 py-3 border-t border-border flex gap-2">
   372→                <Button
   373→                  size="sm"
   374→                  className="flex-1"
   375→                  onClick={() => saveSection(selectedSectionData.id)}
   376→                  disabled={saving === selectedSectionData.id}
   377→                >
   378→                  <Save className="h-3.5 w-3.5 mr-1" />
   379→                  {saving === selectedSectionData.id ? 'Saving...' : 'Save'}
   380→                </Button>
   381→              </div>
   382→            </Card>
   383→          </div>
   384→        </div>
   385→      )}
   386→    </div>
   387→  )
   388→}

   390→// ============ Preview components (mini live render) ============
   391→function SectionPreview({ section }) {
   392→  const config = typeof section.config === 'string' ? JSON.parse(section.config) : section.config
   393→
   394→  switch (section.sectionType) {
   395→    case 'hero':
   396→      return (
   397→        <div className="relative h-32 bg-gradient-to-br from-muted to-muted/60 overflow-hidden">
   398→          {config.image && <img src={config.image} alt="" className="absolute inset-0 h-full w-full object-cover opacity-40" />}
   399→          <div className="absolute inset-0 flex items-center justify-center text-center p-4">
   400→            <div>
   401→              <p className="text-sm font-bold truncate">{config.title || 'Hero Banner'}</p>
   402→              <p className="text-xs text-muted-foreground truncate mt-0.5">{config.subtitle}</p>
   403→              {config.buttonText && <span className="inline-block mt-1 text-[10px] px-2 py-0.5 rounded bg-foreground text-background">{config.buttonText}</span>}
   404→            </div>
   405→          </div>
   406→        </div>
   407→      )
   408→
   409→    case 'product-grid':
   410→      return (
   411→        <div className="p-4">
   412→          <p className="text-sm font-semibold mb-2">{config.title || 'Product Grid'}</p>
   413→          <div className="grid grid-cols-4 gap-2">
   414→            {Array.from({ length: config.limit || 4 }).slice(0, 4).map((_, i) => (
   415→              <div key={i} className="aspect-square rounded bg-muted/50 flex items-center justify-center">
   416→                <Grid3x3 className="h-4 w-4 text-muted-foreground/50" />
   417→              </div>
   418→            ))}
   419→          </div>
   420→          <p className="text-[10px] text-muted-foreground mt-2">Source: {config.source} · {config.limit || 8} products</p>
   421→        </div>
   422→      )
   423→
   424→    case 'marquee':
   425→      return (
   426→        <div className="py-2 overflow-hidden" style={{ backgroundColor: config.bg || '#1a1a1a' }}>
   427→          <p className="text-xs font-medium text-center px-4 truncate" style={{ color: config.color || '#fff' }}>
   428→            {config.text || 'Marquee text...'}
   429→          </p>
   430→        </div>
   431→      )
   432→
   433→    case 'text':
   434→      return (
   435→        <div className={`p-4 text-${config.align || 'center'}`}>
   436→          <p className="text-sm font-semibold">{config.title || 'Text Block'}</p>
   437→          <p className="text-xs text-muted-foreground mt-1 line-clamp-2">{config.body?.replace(/<[^>]+>/g, '')}</p>
   438→        </div>
   439→      )
   440→
   441→    case 'image':
   442→      return (
   443→        <div className="p-4">
   444→          {config.url ? (
   445→            <img src={config.url} alt={config.alt || ''} className="w-full h-24 object-cover rounded" />
   446→          ) : (
   447→            <div className="w-full h-24 rounded bg-muted/50 flex items-center justify-center">
   448→              <ImageIcon className="h-6 w-6 text-muted-foreground/50" />
   449→            </div>
   450→          )}
   451→        </div>
   452→      )
   453→
   454→    case 'newsletter':
   455→      return (
   456→        <div className="p-4 bg-foreground text-background text-center">
   457→          <p className="text-sm font-bold">{config.title || 'Newsletter'}</p>
   458→          <p className="text-xs opacity-80 mt-0.5">{config.subtitle}</p>
   459→          <div className="mt-2 inline-flex gap-1">
   460→            <div className="h-6 w-24 rounded bg-background/20" />
   461→            <div className="h-6 w-12 rounded bg-background" />
   462→          </div>
   463→        </div>
   464→      )
   465→
   466→    case 'discounts':
   467→      return (
   468→        <div className="p-4">
   469→          <p className="text-sm font-semibold mb-2">{config.title || 'Discount Codes'}</p>
   470→          <div className="flex gap-2">
   471→            {[1, 2].map(i => (
   472→              <div key={i} className="flex-1 p-2 rounded border-2 border-dashed border-border">
   473→                <div className="h-2 w-12 bg-muted rounded mb-1" />
   474→                <div className="h-3 w-16 bg-muted rounded" />
   475→              </div>
   476→            ))}
   477→          </div>
   478→        </div>
   479→      )
   480→
   481→    case 'custom-html':
   482→      return (
   483→        <div className="p-4">
   484→          <div className="font-mono text-xs text-muted-foreground bg-muted/30 p-2 rounded line-clamp-3">
   485→            {config.html || '<!-- Custom HTML -->'}
   486→          </div>
   487→        </div>
   488→      )
   489→
   490→    default:
   491→      return <div className="p-4 text-xs text-muted-foreground">Unknown section type</div>
   492→  }
   493→}
   494→
   495→function defaultConfig(type) {
   496→  const defaults = {
   497→    hero: { title: 'New Section', subtitle: 'Add your subtitle here', buttonText: 'Shop Now', buttonUrl: '/products', image: '', overlay: 'dark' },
   498→    'product-grid': { title: 'Products', source: 'featured', limit: 8 },
